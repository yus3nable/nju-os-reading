```
void wait_for_beat(int current_beat) {
retry:
    mutex_lock(&lk);
    int conductor_beat_ = conductor_beat;
    mutex_unlock(&lk);
    if (current_beat >= conductor_beat_) {
        goto retry;
    }
}

void release_beat() {
    mutex_lock(&lk);
    conductor_beat++;
    mutex_unlock(&lk);
}
```
