```
void T_producer() {
    mutex_lock(&lk);
    while (!(depth < n)) {
        cond_wait(&cv, &lk);
    }
    assert(depth < n);
    depth++;
    printf("(");
    cond_broadcast(&cv);
    mutex_unlock(&lk);
}
```
