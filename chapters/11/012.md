wheel\.c 中的 usleep 覆盖：

```
int usleep(useconds_t usec)
{
    static int (*real_usleep)(useconds_t) = NULL;
    if (!real_usleep) {
        real_usleep = dlsym(RTLD_NEXT, "usleep");
    }
    // 用更短的真实usleep来模拟虚拟加速
    useconds_t real_usec = (useconds_t)((double)usec / speedup);
    return real_usleep(real_usec);
}
```
