```
__global__ void mandelbrot_kernel(int *pic_d) {
  int x = blockIdx.x * blockDim.x + threadIdx.x;
  int y = blockIdx.y * blockDim.y + threadIdx.y;
  if (x < W && y < H) {
    int n = mandelbrot(x, y);
    pic_d[y * W + x] = n;
  }
}
```
