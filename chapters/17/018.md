课程线程库上的全局变量、T\_sum 和 main：

```
#define N 1000000
long sum = 0;
mutex_t lk = MUTEX_INIT();

void T_sum() {
  for (int i = 0; i < N; i++) {
    mutex_lock(&lk);
    sum++;
    mutex_unlock(&lk);

    if (i % (N / 10) == 0) {
      sum++;
    }
  }
}

int main() {
  spawn(T_sum);
  spawn(T_sum);
  join();
  printf("sum = %ld\n", sum);
  printf("2*n = %ld\n", 2L * N);
}
```
