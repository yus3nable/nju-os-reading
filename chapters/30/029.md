［AI 原题的条件变量代码；mutex\_t、cond\_t 及其 API 依赖题目约定。］

```
mutex_t lk = MUTEX_INIT();
cond_t cv = COND_INIT();
int food_ready = 0;

void Philosopher(int id) {
  mutex_lock(&lk);
  while (!food_ready) {
    cond_wait(&cv, &lk);
  }
  printf("Philosopher %d is eating!\n", id);
  mutex_unlock(&lk);
}

void Waiter() {
  mutex_lock(&lk);
  food_ready = 1;
  cond_broadcast(&cv);
  mutex_unlock(&lk);
}
```
