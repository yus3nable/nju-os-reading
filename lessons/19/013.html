static const char *coro_resume(struct coroutine *c) {
  if (setjmp(c->ctx) == 0) {
    longjmp(c->coro_ctx, 1);
  }
  return c->buf;
}

static void coro_entry(struct coroutine *c) {
  while (1) {
    c->i++;
    snprintf(c->buf, sizeof(c->buf), "[%d] i = %d", c->name, c->i);
    coro_yield(c);
  }
}
