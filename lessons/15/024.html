void T_worker(int tid) {
    tid--;
    mutex_lock(&mutex);
    while (!(tasks[tid].pending_deps == 0)) {
        cond_wait(&tasks[tid].cv, &mutex);
    }
    mutex_unlock(&mutex);

    printf("Computing node #%d...\n", tid);
    sleep(1);
    // 后面更新后继任务的依赖。
}

mutex_lock(&mutex);
for (int i = 0; i < LENGTH(edges); i++) {
    struct task *t = &tasks[edges[i].dst];
