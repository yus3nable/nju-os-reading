signal.c 片段：头文件、handler 和 cleanup：

#include <stdio.h>
#include <stdlib.h>
#include <signal.h>
#include <unistd.h>

void handler(int signum) {
    switch (signum) {
        case SIGINT:
            printf("Received SIGINT!\n");
            break;
        case SIGQUIT:
            printf("Received SIGQUIT!\n");
            exit(0);
            break;
    }
}

void cleanup() {
    printf("atexit() cleanup\n");
}

        buf[nread - 1] = '\0';
        printf("[%d] Got: %s\n", getpid(), buf);

            perror("read");
            exit(1);

}
